    58	export function reconcileTabs(m: SeatRowsModel, prev: TabState | undefined, on: boolean): { state: TabState; view: TabView } {
    59	  const p = prev ?? EMPTY;
    60	  if (!on) return { state: { ...p, zoom: null }, view: { tabs: [], active: null, zoom: null, hidden: [] } };
    61	  const groups: TabInfo[] = m.groups.map((g) => ({ key: g.key, label: g.label, sids: g.rows.map((r) => r.sid) }));
    62	  const keys = groups.map((g) => g.key);
    63	  // 순서: 저장 순서 중 지금 있는 것 → 새 묶음은 모델 순(처음 나타난 순)으로 뒤에.
    64	  const order = [...p.order.filter((k) => keys.includes(k)), ...keys.filter((k) => !p.order.includes(k))];
    65	  const zoomGroup = p.zoom === null ? undefined : groups.find((g) => g.sids.includes(p.zoom as number));
    66	  const zoom = zoomGroup ? p.zoom : null;
    67	  let active: string | null = zoomGroup ? zoomGroup.key : p.active !== null && keys.includes(p.active) ? p.active : (order[0] ?? null);
    68	  const all = groups.flatMap((g) => g.sids);
    69	  const multi = groups.length > 1;
    70	  let hidden: number[];
    71	  if (zoom !== null) hidden = all.filter((s) => s !== zoom);
    72	  else if (multi) hidden = all.filter((s) => !groups.find((g) => g.key === active)!.sids.includes(s));
    73	  else hidden = [];
    74	  if (!multi && zoom === null) active = p.active; // 탭이 없을 땐 저장 활성값을 건드리지 않는다(묶음이 다시 늘면 그대로)

    77	    state: { order: multi ? order : [...p.order, ...keys.filter((k) => !p.order.includes(k))], active, zoom },

    79	  };
    80	}
