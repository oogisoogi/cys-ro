맥락: 터미널 앱(xterm.js 5.5 · Tauri UI)에서 좌석 프로세스가 끝나면 pane 에 빨간 「[surface exited]」 배너를 쓴다. 종전 코드는 `term.write("\r\n\x1b[31m[surface exited]\x1b[0m\r\n")` 로 **커서 다음 줄**에 써서, 앱이 커서를 화면 중간(입력 상자 · 대체 화면 TUI)에 둔 채 끝나면 배너가 중간에 뜨고 그 아래 줄을 덮어썼다(헤드리스 재현 · 대조군 확인).
수리: 앞선 쓰기(필터 잔여 방류 · 마우스 모드 DECRST reset)가 해석된 뒤(write 콜백) 활성 버퍼에서 「커서 줄과 그 아래 마지막 비어 있지 않은 줄 중 더 아래」를 찾아 `ESC[r`(스크롤 영역 해제) + CUP 로 옮긴 다음 종전 배너를 쓴다. 대체 화면(1049)은 일부러 풀지 않는다(죽은 TUI 마지막 화면 보존). 종료 이벤트 뒤 같은 터미널에 출력은 다시 오지 않는다(스트림은 pane 생성 시 1회만 시작 · pane 생성은 멱등). 단, 이 종료 이벤트는 데몬 연결 끊김(EOF)에도 발화한다.

+export type ScreenView = { rows: number; cursorY: number; line: (y: number) => string };
+
+/// 배너 바로 앞 줄(0-기준 화면 행) — 커서 줄과 「그 아래 마지막 내용 줄」 중 더 아래.
+/// 커서 아래에 내용이 없으면 커서 줄 그대로(종전과 같은 자리 — 멀쩡한 화면은 바뀌지 않는다).
+export function bannerRow(v: ScreenView): number {
+  for (let y = v.rows - 1; y > v.cursorY; y--) {
+    if (v.line(y).trim() !== "") return y;
+  }
+  return v.cursorY;
+}

+export function exitedBannerSeq(v: ScreenView): string {
+  return `${RESET_MARGINS}\x1b[${bannerRow(v) + 1};1H${EXITED_BANNER}`;
+}

+type Line = { translateToString(trimRight?: boolean): string };
+export type ExitTerm = {
+  rows: number;
+  buffer: { active: { baseY: number; cursorY: number; getLine(y: number): Line | undefined } };
+  write(data: string | Uint8Array, callback?: () => void): void;
+};

+/// 종료 쓰기 순서(스펙 D4 ③ 유지): ①필터 잔여 carry 방류 ②정합기 reset(필터 우회 직접 기록) ③배너.
+/// ③의 위치 판독은 ①②와 앞서 대기열에 쌓인 출력이 **해석된 뒤**(write 콜백)에 한다 — 먼저 읽으면
+/// 아직 그려지지 않은 화면을 보고 틀린 줄을 고른다. 판독이 실패해도(창 파괴 등) 배너는 1회 쓴다.
+export function writeExitedBanner(term: ExitTerm, filter: ExitFilter, done?: () => void): void {
+  const rest = filter.flush();
+  if (rest.length > 0) term.write(rest);
+  term.write(filter.reset(), () => {
+    let seq = EXITED_BANNER;
+    try {
+      const buf = term.buffer.active;
+      seq = exitedBannerSeq({ rows: term.rows, cursorY: buf.cursorY, line: (y) => buf.getLine(buf.baseY + y)?.translateToString(true) ?? "" });
+    } catch {
+      /* 판독 실패 — 종전 자리(커서 다음 줄)로 강등 */
+    }
+    term.write(seq, done);
+  });
+}
